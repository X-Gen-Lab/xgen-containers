cmake_minimum_required(VERSION 3.24)
project(xgen_containers VERSION 0.1.0 DESCRIPTION "Bounded non-owning C11 containers" LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/xgenContainersCompatibility.cmake)
option(XGCT_BUILD_LIST "Build intrusive doubly-linked lists" ON)
option(XGCT_BUILD_HASH "Build fixed-capacity integer-key hashes" ON)
option(XGCT_BUILD_BITSET "Build caller-owned bitsets" ON)
option(XGCT_BUILD_RING_BUFFER "Build byte rings and zero-copy claims" ON)
option(XGCT_BUILD_TESTS "Build host GoogleTest and consumers" OFF)
option(XGCT_ENABLE_COVERAGE "Instrument production C objects" OFF)
set(XGCT_STATUS_SOURCE_DIR "" CACHE PATH "Explicit optional xgen-status source")
set(XGCT_COMPONENTS "")
foreach(component list hash bitset ring_buffer)
    string(TOUPPER "${component}" upper)
    if(XGCT_BUILD_${upper})
        list(APPEND XGCT_COMPONENTS ${component})
    elseif(XGCT_BUILD_TESTS)
        message(FATAL_ERROR "Tests require all containers")
    endif()
endforeach()
if(NOT XGCT_COMPONENTS)
    message(FATAL_ERROR "Select at least one container")
endif()
set(existing FALSE)
foreach(component list hash bitset ring_buffer)
    if(TARGET xgct::${component})
        set(existing TRUE)
    endif()
endforeach()
if(existing)
    foreach(component IN LISTS XGCT_COMPONENTS)
        if(NOT TARGET xgct::${component})
            message(FATAL_ERROR "Incomplete supplied target set: xgct::${component}")
        endif()
        xgct_check_target(xgct::${component} STATIC_LIBRARY "${PROJECT_VERSION}" error)
        if(error)
            message(FATAL_ERROR "${error}")
        endif()
    endforeach()
    return()
endif()
if(XGCT_BUILD_HASH OR XGCT_BUILD_BITSET OR XGCT_BUILD_RING_BUFFER)
    if(NOT TARGET xgs::status)
        if(XGCT_STATUS_SOURCE_DIR)
            add_subdirectory("${XGCT_STATUS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/status")
        else()
            find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS status)
        endif()
    endif()
    get_target_property(status_version xgs::status XGS_VERSION)
    get_target_property(status_abi xgs::status XGS_ABI_VERSION)
    get_target_property(status_type xgs::status TYPE)
    if(NOT status_version VERSION_EQUAL "0.1.0" OR NOT status_abi STREQUAL "1" OR
       NOT status_type STREQUAL "INTERFACE_LIBRARY")
        message(FATAL_ERROR "Incompatible supplied xgs::status version/ABI/type")
    endif()
endif()
if(XGCT_ENABLE_COVERAGE AND (CMAKE_CROSSCOMPILING OR NOT XGCT_BUILD_TESTS OR
   NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$" OR
   CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC"))
    message(FATAL_ERROR "Coverage requires GCC-compatible host tests")
endif()
configure_file(include/xgen/containers/version.h.in generated/xgen/containers/version.h @ONLY)
function(xgct_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow
            $<$<COMPILE_LANGUAGE:C>:-Wstrict-prototypes>)
    endif()
endfunction()
foreach(component IN LISTS XGCT_COMPONENTS)
    add_library(xgct_${component} STATIC src/${component}.c)
    add_library(xgct::${component} ALIAS xgct_${component})
    set_target_properties(xgct_${component} PROPERTIES EXPORT_NAME ${component}
        C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
        XGCT_VERSION ${PROJECT_VERSION} XGCT_ABI_VERSION 1
        EXPORT_PROPERTIES "XGCT_VERSION;XGCT_ABI_VERSION")
    target_compile_features(xgct_${component} PUBLIC c_std_11)
    target_include_directories(xgct_${component} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    if(NOT component STREQUAL "list")
        target_link_libraries(xgct_${component} PUBLIC xgs::status)
    endif()
    xgct_warnings(xgct_${component})
    if(XGCT_ENABLE_COVERAGE)
        target_compile_options(xgct_${component} PRIVATE --coverage -O0 -g)
        target_link_options(xgct_${component} INTERFACE --coverage)
    endif()
    install(TARGETS xgct_${component} EXPORT xgenContainers${component}Targets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(EXPORT xgenContainers${component}Targets
        FILE xgenContainers-${component}-targets.cmake NAMESPACE xgct::
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgen_containers)
    install(FILES include/xgen/containers/${component}.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/xgen/containers)
endforeach()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/xgen/containers/version.h"
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/xgen/containers)
configure_package_config_file(cmake/xgen_containersConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/xgen_containersConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgen_containers)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/xgen_containersConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/xgen_containersConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/xgen_containersConfigVersion.cmake"
    cmake/xgenContainersCompatibility.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/xgen_containers)
if(XGCT_BUILD_TESTS)
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Host tests cannot run in cross builds")
    endif()
    enable_language(CXX)
    find_package(GTest 1.16.0 EXACT CONFIG REQUIRED)
    enable_testing()
    add_subdirectory(tests)
endif()
